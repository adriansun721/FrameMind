cmake_minimum_required(VERSION 3.20)
project(framemind_native VERSION 0.1.0 LANGUAGES CXX)
add_executable(framemind-core src/main.cpp)
target_compile_features(framemind-core PRIVATE cxx_std_17)
target_include_directories(framemind-core PRIVATE .deps/eigen-3.4.0 .deps)
target_compile_definitions(framemind-core PRIVATE EIGEN_MPL2_ONLY EIGEN_DONT_PARALLELIZE)
if(MSVC)
  set_property(TARGET framemind-core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(framemind-core PRIVATE /W4 /utf-8 /fp:precise)
endif()
